La configuración web no es secreta. Las reglas de Firestore son las que protegen los datos.

Reglas de seguridad de Firestore 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }
    function myProfile() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }
    function active() {
      return signedIn() && myProfile().active == true;
    }
    function admin() {
      return active() && myProfile().role == 'admin';
    }

    match /users/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || admin());
      allow create: if signedIn()
        && request.auth.uid == uid
        && request.auth.token.email == 'alfredognzlridyard@gmail.com'
        && request.resource.data.role == 'admin'
        && request.resource.data.active == true;
      allow create: if admin();
      allow update: if admin()
        || (signedIn()
            && request.auth.uid == uid
            && request.resource.data.role == resource.data.role
            && request.resource.data.active == resource.data.active);
      allow delete: if admin();
    }

    match /lonas/{id} {
      allow read: if true;
      allow write: if active();
    }
  }
}
